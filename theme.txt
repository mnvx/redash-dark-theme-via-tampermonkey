  let enabled = GM_getValue('enabled', true);
  let styleEl = null;

  function apply() {
    if (enabled && !styleEl) {

      if (styleEl) styleEl.id = STYLE_ID;
    } else if (!enabled && styleEl) {
      styleEl.remove();
      styleEl = null;

      function toggle() {
    enabled = !enabled;
    GM_setValue('enabled', enabled);
    apply();
    if (enabled) fixWhites(); else unfixWhites();
  }

  apply();
  GM_registerMenuCommand('Toggle Redash dark', toggle);

  document.addEventListener('keydown', (e) => {
    if (e.altKey && e.shiftKey && (e.code === 'KeyD')) {
      e.preventDefault();
      toggle();

      // ---------- Fallback: перекрашиваем всё, что осталось белым ----------
  const SKIP = 'img, svg, canvas, video, iframe, .ace_editor, .ace_editor *, ' +
               '.js-plotly-plot *, .leaflet-container *, .ant-btn-primary, .btn-primary';
  const SURFACE = '#1b222b';

  function isWhite(bg) {
    const m = bg && bg.match(/rgba?\(([\d.]+),\s*([\d.]+),\s*([\d.]+)(?:,\s*([\d.]+))?\)/);
    if (!m) return false;
    const [r, g, b] = [+m[1], +m[2], +m[3]];
    const a = m[4] === undefined ? 1 : +m[4];
    return a > 0.5 && r >= 235 && g >= 235 && b >= 235;
  }

  function fixWhites(root) {
    if (!enabled || !document.body) return;
    const nodes = (root && root.querySelectorAll ? root : document.body).querySelectorAll('*');
    for (const el of nodes) {
      if (el.matches(SKIP)) continue;
      if (isWhite(getComputedStyle(el).backgroundColor)) {
        el.style.setProperty('background-color', SURFACE, 'important');
        el.classList.add('rd-dark-fixed');
      }
    }
  }

  function unfixWhites() {
    document.querySelectorAll('.rd-dark-fixed').forEach((el) => {
      el.style.removeProperty('background-color');
      el.classList.remove('rd-dark-fixed');
    });
  }

  let fixTimer = null;
  const observer = new MutationObserver(() => {
    clearTimeout(fixTimer);
    fixTimer = setTimeout(() => fixWhites(), 150);
  });

  function startFixer() {
    fixWhites();
    observer.observe(document.body, { childList: true, subtree: true, attributes: true, attributeFilter: ['class', 'style'] });
  }

  if (document.body) startFixer();
  else document.addEventListener('DOMContentLoaded', startFixer);
